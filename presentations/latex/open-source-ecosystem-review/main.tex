\begin{frame}{O que o artigo faz bem}

\begin{frame}{O que o artigo não prova}

  \begin{columns}[T,onlytextwidth]
    \column{0.33\textwidth}
      \begin{center}\boxword{softred}{causalidade}\end{center}
      \centering\small O modelo não é testado como causa de crescimento.
    \column{0.33\textwidth}
      \begin{center}\boxword{softred}{generalização}\end{center}
